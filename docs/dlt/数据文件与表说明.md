# 数据表说明

## 数据处理

### dlt_draws

体彩大乐透开奖结果存储文件 `data/processed/dlt/draws.parquet`

#### SQL 语句

```sql
-- 创建语句

CREATE TABLE dlt_draws (
-- =========================================================================
-- 基础信息
-- =========================================================================

-- 开奖期号，例如：26107
issue VARCHAR NOT NULL,

-- 开奖日期，例如：2026-09-19
draw_date DATE NOT NULL,

-- 官方开奖字符串，例如：
-- "02 05 07 14 22 04 10"
-- 前五个号码为前区，后两个号码为后区
draw_result VARCHAR NOT NULL,

-- =========================================================================
-- 前区号码
-- 大乐透规则：01-35 选择 5 个号码
-- =========================================================================

-- 前区第 1 个号码（升序）
red_1 INTEGER NOT NULL,

-- 前区第 2 个号码（升序）
red_2 INTEGER NOT NULL,

-- 前区第 3 个号码（升序）
red_3 INTEGER NOT NULL,

-- 前区第 4 个号码（升序）
red_4 INTEGER NOT NULL,

-- 前区第 5 个号码（升序）
red_5 INTEGER NOT NULL,

-- =========================================================================
-- 后区号码
-- 大乐透规则：01-12 选择 2 个号码
-- =========================================================================

-- 后区第 1 个号码（升序）
blue_1 INTEGER NOT NULL,

-- 后区第 2 个号码（升序）
blue_2 INTEGER NOT NULL,

-- =========================================================================
-- 基础统计字段
-- 不依赖历史数据，可由开奖号码直接计算
-- =========================================================================

-- 前区号码和值：
-- red_1 + red_2 + red_3 + red_4 + red_5
red_sum INTEGER NOT NULL,

-- 前区跨度：
-- red_5 - red_1
red_span INTEGER NOT NULL,

-- 后区号码和值：
-- blue_1 + blue_2
blue_sum INTEGER NOT NULL,

-- 后区跨度：
-- blue_2 - blue_1
blue_span INTEGER NOT NULL,

-- =========================================================================
-- 奇偶统计
-- =========================================================================

-- 前区奇数号码数量
red_odd_count INTEGER NOT NULL,

-- 前区偶数号码数量
red_even_count INTEGER NOT NULL,

-- 后区奇数号码数量
blue_odd_count INTEGER NOT NULL,

-- 后区偶数号码数量
blue_even_count INTEGER NOT NULL,

-- =========================================================================
-- 前区区间分布
--
-- 分区：
-- zone_1: 01-12
-- zone_2: 13-24
-- zone_3: 25-35
-- =========================================================================

-- 前区号码落入 01-12 区间的数量
red_zone_1_count INTEGER NOT NULL,

-- 前区号码落入 13-24 区间的数量
red_zone_2_count INTEGER NOT NULL,

-- 前区号码落入 25-35 区间的数量
red_zone_3_count INTEGER NOT NULL,

-- =========================================================================
-- 连号统计
-- =========================================================================

-- 前区连续号码数量统计。
--
-- 例如：
-- 02 03 10 20 22
--
-- 存在一组连续号码：
-- 02-03
--
-- 则值为 1
red_consecutive_count INTEGER NOT NULL,

-- =========================================================================
-- 时间特征
-- =========================================================================

-- 开奖年份
year INTEGER NOT NULL,

-- 开奖月份
month INTEGER NOT NULL,

-- 开奖星期。
--
-- ISO 标准：
-- Monday=1 ... Sunday=7
day_of_week INTEGER NOT NULL,

-- =========================================================================
-- 数据索引
-- =========================================================================

-- 数据集内部索引。
--
-- 当前约定：
-- 最新一期 = 0
-- 越早的开奖值越大
draw_index INTEGER NOT NULL
);
```

```sql
-- 前区约束

ALTER TABLE dlt_draws ADD CONSTRAINT check_red_range CHECK ( red_1 BETWEEN 1 AND 35 AND red_2 BETWEEN 1 AND 35 AND red_3 BETWEEN 1 AND 35 AND red_4 BETWEEN 1 AND 35 AND red_5 BETWEEN 1 AND 35 );
```

```sql
--- 后区约束

ALTER TABLE dlt_draws ADD CONSTRAINT check_blue_range CHECK ( blue_1 BETWEEN 1 AND 12 AND blue_2 BETWEEN 1 AND 12 );
```
